const json = pm.response.json();

pm.test('TC01 - Status is 200', function () {
  pm.response.to.have.status(200);
});

pm.test('TC01 - Response contains token string', function () {
  pm.expect(json).to.have.property('token');
  pm.expect(json.token).to.be.a('string').and.to.have.length.above(20);
});

pm.test('TC01 - empCode in response matches request', function () {
  pm.expect(json.empCode).to.equal(pm.environment.get('empCode'));
});

pm.test('TC01 - username is a non-empty string', function () {
  pm.expect(json.username).to.be.a('string').and.to.have.length.above(0);
});

pm.test('TC01 - roles is a non-empty array', function () {
  pm.expect(json.roles).to.be.an('array').and.to.have.length.above(0);
});

pm.test('TC01 - Response time under 3000ms', function () {
  pm.expect(pm.response.responseTime).to.be.below(3000);
});

if (json.token) {
  pm.environment.set('authToken', json.token);
  console.log('authToken stored — length: ' + json.token.length);
}
