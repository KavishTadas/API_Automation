const json = pm.response.json();

pm.test('TC02 - Status is 400 or 401 for wrong password', function () {
  pm.expect(pm.response.code).to.be.oneOf([400, 401]);
});

pm.test('TC02 - Response body has message field', function () {
  pm.expect(json).to.have.property('message');
  pm.expect(json.message).to.be.a('string').and.to.have.length.above(0);
});
